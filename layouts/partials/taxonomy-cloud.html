{{- /* .ByCount only to find the displayed range; the render loop below stays alphabetical. */ -}} {{- $eligible := where (index site.Taxonomies $taxonomy).ByCount "Count" "ge" $minCount }} {{- if $eligible }} {{- $largestFontSize := 3.8 }} {{- $smallestFontSize := 0.8 }} {{- /* Scale across the *displayed* terms. Anchoring $min to the full taxonomy's minimum (1) while only showing 2-and-up would strand the bottom of the range on chips that are never drawn, pushing every surviving chip up and flattening the size variety. The +1 on $max is load-bearing, not cosmetic: it guarantees log($max) > log($min) even when every shown term has the same count, which is what keeps the divisor below non-zero. It also removes the theme's latent divide-by-zero when a taxonomy has one term. */ -}} {{- $max := add (index $eligible 0).Count 1 }} {{- $min := (index $eligible (sub (len $eligible) 1)).Count }} {{- $logSpread := sub (math.Log $max) (math.Log $min) }} {{- range $name, $taxonomyPage := (index site.Taxonomies $taxonomy) }} {{- $tagCount := len $taxonomyPage.Pages }} {{- if ge $tagCount $minCount }} {{- $weight := div (sub (math.Log $tagCount) (math.Log $min)) $logSpread }} {{- $fontSize := add $smallestFontSize (mul (sub $largestFontSize $smallestFontSize) $weight) }} {{- /* The lime gradient stays: it reads as brand-alt, and it is the chip's fill.
