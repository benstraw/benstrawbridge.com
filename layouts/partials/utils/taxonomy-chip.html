{{- /* One tag-cloud chip. Extracted from the theme's taxonomy-cloud.html so the head (sized, by count) and tail (uniform, alphabetical) loops in the site override don't each carry a copy of the gradient/twClasses resolution. Params: taxonomy string plural taxonomy name, e.g. "musical-genres" name string term name as it appears in site.Taxonomies count int pages carrying the term fontSize float rem value for the inline font-size compact bool optional; tighter padding, for the collapsed tail page Page the page being rendered, for the Kind check below Kept deliberately identical to the theme's markup: same .tagcloud-item hook, same chip classes, same count. Only the padding varies, and only when compact is set. */ -}} {{- $taxonomy := .taxonomy }} {{- $name := .name }} {{- $count := .count }} {{- $fontSize := .fontSize }} {{- $page := .page }} {{- /* The lime gradient stays: it reads as brand-alt, and it is the chip's fill. Only the border/hover/ring move to accent. See the theme partial and docs/migration/v0.4.0.md for why. */ -}} {{- $gradientClass := "bg-gradient-to-r from-lime-400 to-lime-500 text-zinc-900 " }} {{- $chipAccent := "border-ryder-accent-500 hover:bg-ryder-accent-500 hover:border-ryder-accent-600 focus:ring-ryder-accent-300" }} {{- if (site.Params.colors | default dict).legacyAccents }} {{- $chipAccent = "border-yellow-500 hover:bg-yellow-500 hover:border-yellow-600 focus:ring-yellow-300" }} {{- end }} {{- $padding := cond .compact "py-1 px-3" "py-2 px-4" }} {{- $classes := printf "%s rounded-full border-2 %s focus:outline-none focus:ring-2 no-underline inline-block text-center break-words " $padding $chipAccent }} {{- with site.Param "twClasses.headerBackgroundFrameOuter" }} {{- $gradientClass = . }} {{- end }} {{- /* Resolve the term's own page once: it supplies both the canonical URL and the optional per-term chip fill. Everything goes through `with` because site.GetPage returns nil for a term whose page Hugo did not build, and reading .Params off that would abort the build. Falling back to a hand-built href keeps a missing term page from dropping the chip. */ -}} {{- $href := printf "%s/%s/" $taxonomy ($name | urlize) | relURL }} {{- with site.GetPage (printf "%s/%s" $taxonomy $name) }} {{- $href = .RelPermalink }} {{- if eq $page.Kind "taxonomy" }} {{- with .Params.twClasses.headerBackgroundFrameOuter }} {{- $gradientClass = . }} {{- end }} {{- end }} {{- end }} {{ $name }}{{ $count }}
