{{- /* SITE OVERRIDE of themes/ryder/layouts/partials/utils/taxonomy-string.html. Why: the theme filters each group by `minCount` and then ranges the taxonomy *map*, which is alphabetical and uncapped. On this site that put ~78 links in the footer of every page, ordered so that "acid jazz" (7 artists) preceded "jazz" (97), and growing without limit as the Spotify data grows. What this does instead: order by page count, apply `minCount` as a floor, then cap at `maxTerms`. Drop `maxTerms` from a group's config to get the theme's behaviour back for that group. Note this is a *reordering and bounding* change, not a size reduction: at the configured caps the footer carries 74 links against the theme's 78. The value is that `jazz` (97) leads instead of `acid jazz` (7), and that the count stays fixed as the Spotify library grows. Reconcile this with the theme partial on any Ryder bump — see CLAUDE.md. */ -}} {{- $taxonomies := site.Params.footer.taxonomies }} {{- /* Only render if taxonomies are explicitly configured */ -}} {{- if $taxonomies }}
{{- range $taxonomies }} {{- $taxonomy := .name }} {{- $title := .title }} {{- $minCount := .minCount | default 1 }} {{- $maxTerms := .maxTerms | default 0 | int }} {{- /* The heading links to the taxonomy's own root page (/tags/, /musical-genres/) when Hugo generated one. site.GetPage rather than a hand-built URL: a site can disable the taxonomy kind in [outputs] or rename its path, and a string-concatenated href would then point at a 404. When there is no such page the heading stays plain text — a heading that looks like a link and isn't is worse than one that doesn't. */ -}} {{- $taxonomyPage := site.GetPage $taxonomy }} {{- /* .ByCount is ordered most-used first, so the cap keeps the terms worth showing rather than whichever ones sort early. */ -}} {{- $terms := (index site.Taxonomies $taxonomy).ByCount }} {{- $eligible := where $terms "Count" "ge" $minCount }} {{- $shown := $eligible }} {{- if and (gt $maxTerms 0) (gt (len $eligible) $maxTerms) }} {{- $shown = first $maxTerms $eligible }} {{- end }}

{{- with $taxonomyPage -}} {{ $title }} {{- else -}} {{ $title }} {{- end -}}

{{- range $shown }} {{ .Name }}{{ .Count }} {{- end }} {{- /* No "see all" link down here: the group heading above is already an anchor to the same taxonomy page, chevron and all. */ -}}
{{- end }}
{{- end }}
